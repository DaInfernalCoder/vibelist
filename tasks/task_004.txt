```jsx
// Example component structure for profile section
const ProfileSection = ({ title, value, onSave }) => {
  const [isEditing, setIsEditing] = useState(false);
  const [currentValue, setCurrentValue] = useState(value);
  const [isLoading, setIsLoading] = useState(false);
  
  const handleSave = async () => {
    setIsLoading(true);
    try {
      await onSave(currentValue);
      toast.success(`${title} updated successfully`);
      setIsEditing(false);
    } catch (error) {
      toast.error(`Failed to update ${title}`);
    } finally {
      setIsLoading(false);
    }
  };
  
  return (
    <div className="card bg-base-100 shadow-sm mb-4 p-4">
      <h3 className="text-lg font-medium mb-2">{title}</h3>
      {isEditing ? (
        <div className="flex flex-col gap-2">
          <input 
            type="text" 
            className="input input-bordered w-full" 
            value={currentValue}
            onChange={(e) => setCurrentValue(e.target.value)}
          />
          <div className="flex gap-2">
            <button 
              className="btn btn-primary btn-sm" 
              onClick={handleSave}
              disabled={isLoading}
            >
              {isLoading ? <span className="loading loading-spinner"></span> : 'Save'}
            </button>
            <button 
              className="btn btn-ghost btn-sm" 
              onClick={() => {
                setIsEditing(false);
                setCurrentValue(value);
              }}
            >
              Cancel
            </button>
          </div>
        </div>
      ) : (
        <div className="flex justify-between items-center">
          <p>{value}</p>
          <button 
            className="btn btn-ghost btn-sm" 
            onClick={() => setIsEditing(true)}
          >
            Edit
          </button>
        </div>
      )}
    </div>
  );
};
```

Password validation implementation:
```javascript
const validatePassword = (password, confirmPassword) => {
  const errors = [];
  
  if (password.length < 8) {
    errors.push("Password must be at least 8 characters");
  }
  
  if (!/[A-Z]/.test(password)) {
    errors.push("Password must contain at least one uppercase letter");
  }
  
  if (!/[a-z]/.test(password)) {
    errors.push("Password must contain at least one lowercase letter");
  }
  
  if (!/[0-9]/.test(password)) {
    errors.push("Password must contain at least one number");
  }
  
  if (password !== confirmPassword) {
    errors.push("Passwords do not match");
  }
  
  return errors;
};
```
